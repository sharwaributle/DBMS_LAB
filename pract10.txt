CREATE OR REPLACE PROCEDURE ndf (n IN NUMBER)
AS
  vfname   emp.first_name%TYPE;
  vlname   emp.last_name%TYPE;
  vsalary  emp.salary%TYPE;
BEGIN
  SELECT first_name, last_name, salary
  INTO vfname, vlname, vsalary
  FROM emp
  WHERE emp_no = n;

  DBMS_OUTPUT.PUT_LINE('Name: ' || vfname || ' ' || vlname ||
                       ', Salary: ' || vsalary);

EXCEPTION
  WHEN NO_DATA_FOUND THEN
    DBMS_OUTPUT.PUT_LINE('No employee found with given EMP_NO');
END;
/

CREATE OR REPLACE PROCEDURE ndf_emp (n IN NUMBER)
AS
  vfname   emp.first_name%TYPE;
  vlname   emp.last_name%TYPE;
  vsalary  emp.salary%TYPE;

  -- Custom exception for multiple rows
  max_rows_found EXCEPTION;

  v_count NUMBER;
BEGIN
  -- Count rows matching EMP_NO
  SELECT COUNT(*)
  INTO v_count
  FROM emp
  WHERE emp_no = n;

  -- Multiple rows → custom exception
  IF v_count > 1 THEN
    RAISE max_rows_found;
  ELSIF v_count = 1 THEN
    -- Exactly 1 row → fetch data
    SELECT first_name, last_name, salary
    INTO vfname, vlname, vsalary
    FROM emp
    WHERE emp_no = n;

    DBMS_OUTPUT.PUT_LINE('Name: ' || vfname || ' ' || vlname ||
                         ', Salary: ' || vsalary);
  ELSE
    -- 0 rows → message
    DBMS_OUTPUT.PUT_LINE('No employee found with EMP_NO ' || n);
  END IF;

EXCEPTION
  WHEN max_rows_found THEN
    DBMS_OUTPUT.PUT_LINE('Multiple employees found with EMP_NO ' || n);
END;
/

CREATE OR REPLACE PROCEDURE UDF (m1 IN NUMBER, m2 IN NUMBER, m3 IN NUMBER)
AS
    TOTAL NUMBER;
    PERCENTAGE NUMBER;
    Invalid_Marks EXCEPTION;
BEGIN
    IF m1 < 0 OR m1 > 100 OR m2 < 0 OR m2 > 100 OR m3 < 0 OR m3 > 100 THEN
        RAISE Invalid_Marks;
        
    ELSIF m1 > 50 AND m2 > 50 AND m3 > 50 THEN
        DBMS_OUTPUT.PUT_LINE('RESULT = PASS');
        
    ELSE
        DBMS_OUTPUT.PUT_LINE('RESULT = FAIL');
    END IF;

EXCEPTION
    WHEN Invalid_Marks THEN 
        DBMS_OUTPUT.PUT_LINE('Enter valid marks for m1, m2, m3 between 0 - 100');
END;
/

RETURN VARCHAR2
AS
  c NUMBER;
BEGIN
  c := n * n;
  RETURN 'SQUARE OF ' || n || ' = ' || c;
END;
